<template>
  <div class="w-full p-2 flex flex-col ">
    <div class="flex flex-row w-full  items-center justify-center  ">
         <!-- formulaire de calcul de prêt -->
        <div class=" bg-white rounded-3xl p-4  border w-[35%] border-stone-200 ">
            <h1 class="text-3xl font-bold mb-4">Simulateur de Prêt</h1>
            <div class="div ">
              <div class="mb-4 flex flex-col ">
                <label for="montant" class=" mb-2 text-md font-bold text-stone-400 ">Montant du prêt :</label>
                <input type="number" id=" montant" v-model.number="montant"
                  class="border focus:ring-4 focus:ring-green-100 p-2 font-semibold text-4xl outline-none border-stone-200 focus:border-green-500 rounded-2xl" />
              </div>
              <div class="mb-4 flex flex-col ">
                <label for="tauxAnnuel" class=" mb-2 text-md font-bold text-stone-400">Taux d'intérêt annuel (%) :</label>
                <input type="number" id=" tauxAnnuel" v-model.number="tauxAnnuel"
                  class="border focus:ring-4 focus:ring-green-100 p-2 font-semibold text-4xl outline-none border-stone-200 focus:border-green-500 rounded-2xl" />
              </div>
              <div class="mb-4 flex flex-col ">
                <label for="dureeMois" class=" mb-2 text-md font-bold text-stone-400">Durée du prêt (mois) :</label>
                <input type="number" id=" dureeMois" v-model.number="dureeMois"
                  class="border focus:ring-4 focus:ring-green-100 p-2 font-semibold text-4xl outline-none border-stone-200 focus:border-green-500 rounded-2xl" />
              </div>
              <div class="mb-4 ">
                <button @click="calculerAmortissementTableau()"
                  class="bg-green-700 text-3xl px-5 text-white p-2 rounded">Calculer</button>
              </div>
            </div>
        </div>
        <div class="w-[5%]"></div>
        <div class="w-[35%] bg-green-700 h-[470px] p-4 rounded-3xl">
          <span class=" text-white uppercase font-semibold"> Mensualité estimé</span>
        </div>
    </div>
    <!-- tableau d'amortissement -->
    <div v-if="tableauAmortissement.length > 0">
      <h2 class="text-2xl font-bold mb-4">Tableau d'amortissement</h2>
      <table class="table-auto border-collapse border border-gray-400 w-full">
        <thead>
          <tr>
            <th class="border border-gray-400 p-2">Échéance</th>
            <th class="border border-gray-400 p-2">Mensualité hors assurance</th>
            <th class="border border-gray-400 p-2">Intérêts</th>
            <th class="border border-gray-400 p-2">Capital remboursé</th>
            <th class="border border-gray-400 p-2">Capital restant dû</th>
            <th class="border border-gray-400 p-2">Cumul des intérêts</th>
          </tr>
        </thead>
        <tbody>
          <tr v-for="(ligne, index) in tableauAmortissement" :key="index">
            <td class="border border-gray-400 p-2">{{ ligne.echeance }}</td>
            <td class="border border-gray-400 p-2">{{ ligne.colonneF_mensualiteHorsAssurance.toFixed(2) }}</td>
            <td class="border border-gray-400 p-2">{{ ligne.colonneG_interets.toFixed(2) }}</td>
            <td class="border border-gray-400 p-2">{{ ligne.colonneI_capitalRembourse.toFixed(2) }}</td>
            <td class="border border-gray-400 p-2">{{ ligne.colonneJ_capitalRestantDu.toFixed(2) }}</td>
            <td class="border border-gray-400 p-2">{{ ligne.colonneK_cumulInterets.toFixed(2) }}</td>
          </tr>
        </tbody>
      </table>
    </div>
    
  </div>
</template>


<script setup>
import { ref, onMounted } from 'vue'
import heroImg from '../assets/hero.png'


const count = ref(0)

const montant = ref(100000) // Montant du prêt
const tauxAnnuel = ref(5) // Taux d'intérêt annuel en pourcentage
const dureeMois = ref(12) // Durée du prêt en mois
var tableauAmortissement = ref([]) // Tableau d'amortissement 
 
function calculerAmortissementTableau() {
  tableauAmortissement.value = calculerAmortissement(
    montant.value,
    tauxAnnuel.value,
    dureeMois.value
  );
}

function calculerAmortissement(montant, tauxAnnuel, dureeMois) {
  const tauxMensuel = tauxAnnuel / 100 / 12;

  // Colonne F — Mensualité hors assurance
  const mensualiteHorsAssurance =
    montant * tauxMensuel /
    (1 - Math.pow(1 + tauxMensuel, -dureeMois));

  let capitalRestantDu = montant;
  let cumulInterets = 0;
  const tableau = [];

  for (let echeance = 1; echeance <= dureeMois; echeance++) {

    // Colonne G — Intérêts
    const interets = capitalRestantDu * tauxMensuel;

    // Colonne I — Capital remboursé
    const capitalRembourse =
      mensualiteHorsAssurance - interets;

    // Colonne J — Capital restant dû
    capitalRestantDu = Math.max(
      0,
      capitalRestantDu - capitalRembourse
    );

    // Colonne K — Cumul des intérêts
    cumulInterets = cumulInterets + interets;

    tableau.push({
      echeance,
      colonneF_mensualiteHorsAssurance: mensualiteHorsAssurance,
      colonneG_interets: interets,
      colonneI_capitalRembourse: capitalRembourse,
      colonneJ_capitalRestantDu: capitalRestantDu,
      colonneK_cumulInterets: cumulInterets
    });
  }
  return tableau;
}



</script>
